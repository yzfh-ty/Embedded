# 字符串

什么是字符串

字符串是用来保存文字数据的一种数据类型。可以认为字符是 char 类型的一维数组（最后一个字符一定是零`'\0'`）。

字符串的字面值

用双引号括起来。内部是文字信息。

```c
"hello world!" // 函数13个字符的字符串。
"abc"  // 长度是4
```

当使用`scanf()` 读取时，不需要取地址符`&`

## 字符的转义

当字符串字面值的内部有 控制字符或 双引号是，可以使用一个反斜杠加特殊字符表示一个字符，叫做字符的转义，如：

```c
\" 表示双引号  
\' 表示单引号  
\n 表示换行  
\t 表示制表符  
\xHH. hh 表示2位16进制的编码，如\x41 表示 'A'
```

## 字符串的保存

使用char 类型的数组来保存字符串。

```c
char text[100];
char text[100] = {'A', 'B'}
char text[100] = "ABC"; //数组长度为 100,其余为0（'\0'）
char text[] = "ABC";  // 自动推断数组长度为 4。
```

printf 输出字符使用 `%s`占位符

scanf 读取字符串也使用 `%s` 占位符

## 字符串函数

[[02_C语言/22_标准库函数#strlen()\|详细介绍]]

```c
// 求字符串长度
size_t strlen(const char *s);
// 复制字符串 src 源地址 dest 目标地址，返回 dest
char *strcpy(char *dest, const char *src);
// 拼接字符串 src 源地址 dest 目标地址 返回 dest
char *strcat(char *dest, const char *src);
// 比较字符串 s1，和 s2 谁大
int strcmp(const char *s1, const char *s2);
char *strncpy(char *dest, const char *src, size_t n);
char *strncat(char *dest, const char *src, size_t n);
```

示例

```c
#include <stdio.h>  
#include <string.h>  
  
int main(int argc, char * argv[]) {  
 char s1[100] = "hello";  
 char s2[100] = "beijing";  
 char s3[100] = "abc";  
 char s4[100] = "acb";  
 char *p;  
  
 printf("s1的长度%ld\n", strlen(s1));  // 5  
 p = strcpy(s4, s1);   // 复制字符串  
  
 // 复制字符串  
 printf("p:%p, p:%s\n", p, p);  
 printf("s4:%p, s4:%s\n", s4, s4);  
 // 拼接字符串  
 p = strcat(s4, s2);  
 printf("p:%p, p:%s\n", p, p);  
 printf("s4:%p, s4:%s\n", s4, s4); // hellobeijing  
 // 字符串比较  
 printf("%d\n", strcmp("abc", "abcd"));  
 printf("%d\n", strncmp("abc", "abcd", 3)); // 0  
 strncpy(s4, "abcd", 3); // 0  
 printf("%s\n", s4);  
 strncat(s3, "ABCD", 2);  // s3 : "abcAB"  
 printf("%s\n", s3);  
}
```

## 字符串的输入和清除输入缓冲区

以下面这个题目为例

循环输入几个学生的姓名，年龄和身高，当提示输入姓名时，直接回车退出，打印你输入的所有学生的姓名，年龄和身高。

```c
#include <stdio.h>

struct student
{

    char name[50];  // 姓名
    int age;        // 年龄
    float height;   // 身高

};

int main(int argc, char *argv[])
{
    // 定义100个学生类型的数组 class1;
    struct student class1[100];

    int stu_count = 0;  // 用于记录学生数;

    for (int i = 0; i < sizeof(class1) / sizeof(class1[0]); i++)
    {
        printf("请输入姓名: ");
        scanf("%s", class1[i].name);
        printf("请输入年龄: ");
        scanf("%d", &class1[i].age);
        printf("请输入身高: ");
        scanf("%f", &class1[i].height);
        stu_count++;
    }

    for (int i = 0; i < stu_count; i++)
    {
        printf("name:%s, age:%d, height:%f\n",
               class1[i].name,
               class1[i].age,
               class1[i].height);
    }
    return 0;

}
```

上面这个代码无法实现输入为空时退出

当使用 `scangf()` 输入字符串时，如果想要输入为空时退出，会出现无论怎么按空格都无法实现的情况

### 下面提供两种解决方案

使用 `getchar()` 挨个字符读取

```c
#include <stdio.h>
#include <string.h>

int main() {
    char input[100];
    int i = 0;

    while (1) {
        printf("请输入内容（输入空字符串退出）：");
        i = 0;

        // 逐字符读取输入
        while ((input[i] = getchar()) != '\n' && input[i] != '\0') {
            if (i >= 99) {
                // 如果输入超过缓冲区大小，跳过多余字符
                while (getchar() != '\n');
                break;
            }
            i++;
        }
        input[i] = '\0'; // 添加字符串终止符

        // 判断是否为空字符串
        if (strlen(input) == 0) {
            printf("输入为空，程序退出！\n");
            break;
        } else {
            printf("您输入的内容是：%s\n", input);
        }
    }

    return 0;
}
```

使用`fgets` 

```c
#include <stdio.h>
#include <string.h>

int main() {
    char input[100];

    while (1) {
        printf("请输入内容（输入空字符串退出）：");
        if (fgets(input, sizeof(input), stdin) == NULL) {
            printf("读取输入失败，程序退出！\n");
            break;
        }

        // 去掉字符串末尾的换行符
        input[strcspn(input, "\n")] = '\0';

        // 判断是否为空字符串
        if (strlen(input) == 0) {
            printf("输入为空，程序退出！\n");
            break;
        } else {
            printf("您输入的内容是：%s\n", input);
        }
    }

    return 0;
}
```

### 清除输入缓冲区

两个 `scanf()` 连续输入，不清楚缓冲区会出现第二个无法输入直接从缓冲区读取的情况

例如：

```c
// 当输入想要输入 'test' 时，不小心输入了 'test 223'
// 如果不清除缓冲区，下一个 'scanf()' 会读到缓冲区的内容

#include <stdio.h>

int main() {
    char char1[100];
    char char2[100];

    printf("请输入内容1：");
    scanf("%99s", char1);
    printf("请输入内容2：");
    scanf("%99s", char2);

    printf("您输入的内容1是：%s\n", char1);
    printf("您输入的内容2是：%s\n", char2);

    return 0;
}
```

结果：

```c
请输入内容1：test 223
请输入内容2：您输入的内容1是：test
您输入的内容2是：223
```

想要解决这个问题，需要清除输入缓冲区，个人推荐使用 `getchar()` 循环解决这个问题

#### 使用`getchar()`循环

```c
#include <stdio.h>

void clear_input_buffer() {
	// 读取并丢弃所有字符，直到换行符或文件结束符
    // 第一种
    int c;
    while ((c = getchar()) != '\n' && c != EOF) ;
    
    // 也可以不定义变量c
    // 第二种
    while (getchar() != '\n') ;

}

int main() {
    char input[100];

    printf("请输入内容：");
    scanf("%99s", input);
    clear_input_buffer(); // 清除缓冲区中的多余字符

    printf("您输入的内容是：%s\n", input);

    return 0;
}
```

#### 使用`scanf()`读取剩余行

```c
#include <stdio.h>

void clear_input_buffer() {
    scanf("%*c"); // 读取并丢弃一个字符（通常是换行符）
}

int main() {
    char input[100];

    printf("请输入内容：");
    scanf("%99s", input);
    clear_input_buffer(); // 清除缓冲区中的多余字符

    printf("您输入的内容是：%s\n", input);

    return 0;
}
```

#### 使用fgets读取剩余行

```c
#include <stdio.h>

void clear_input_buffer() {
    char buffer[100];
    fgets(buffer, sizeof(buffer), stdin); // 读取并丢弃剩余的整行输入
}

int main() {
    char input[100];

    printf("请输入内容：");
    scanf("%99s", input);
    clear_input_buffer(); // 清除缓冲区中的多余字符

    printf("您输入的内容是：%s\n", input);

    return 0;
}
```
