# 结构体

什么是结构体：

结构体(struct)是用于将不同的类型的变量组合成一个逻辑单元的复合类型。结构体的作用是封装相关联的数据成为一个整体，有利于数据管理。

使代码模块化，增强可读性和易用性。

## 结构体类型定义的语法:

```c
struct 结构体名 {  
    成员类型1  变量名1;  
    成员类型2  变量名2;  
    ...  
};
```

示例

```c
struct student {  
   char name[50];  // 姓名  
   int age;   // 年龄  
   float height; // 身高  
};

// 也可以使用typedef
typedef struct {
	char name[50];  // 姓名  
	int age;        // 年龄  
	float height;   // 身高
} stu;
```

## 结构类型变量的初始化

```c
struct 结构体名 变量名1 = {成员1的初始值,成员2的初始值,...};  
  
struct 结构体名 变量名2 = {.成员变量名1 = 初始值1, .成员变量名2 = 初始值2,...}; //C99及以上
```

示例:

```c
// 定义一个学生类型的变量zhang3,  
struct student zhang3 = {"zhangsan", 18, 1.80 };  
struct student li4 = { .name="lisi", .age=16};   
// 定义100个学生类型的数组 class1;  
struct student class1[100];
```

综合示例

```c
#include <stdio.h>  
  
struct student {  
   char name[50];  // 姓名  
   int age;   // 年龄  
   float height; // 身高  
};  
  
int main(int argc, char * argv[]) {  
 // 定义一个学生类型的变量zhang3,  
 struct student zhang3 = {"zhangsan", 18, 1.80 };  
 struct student li4 = { .name="lisi", .age=16};  
 // 定义100个学生类型的数组 class1;  
 struct student class1[100]; //class1[0]  
  
 printf("name:%s, age:%d, height:%f\n", zhang3.name, zhang3.age, zhang3.height);  
 printf("name:%s, age:%d, height:%f\n", li4.name, li4.age, li4.height);  
 printf("sizeof(zhang3):%ld\n", sizeof(zhang3));  
  
  
 return 0;  
}
```

## 结构成员访问运算符`.`

作用：访问结构体成员。

**语法**

```
结构体变量.成员变量
```

练习：

循环输入几个学生的姓名，年龄和身高，当输入年龄输入负数时结束输入，打印你输入的所有学生的姓名，年龄和身高。

参考答案

```c
#include <stdio.h>  
  
struct student {  
   char name[50];  // 姓名  
   int age;   // 年龄  
   float height; // 身高  
};  
  
int main(int argc, char * argv[]) {  
 // 定义100个学生类型的数组 class1;  
 struct student class1[100]; //class1[0]  
    int stu_count = 0;  // 用于记录学生数;  
  
 for (int i = 0; i < sizeof(class1) /sizeof(class1[0]); i++){  
  printf("请输入姓名: ");  
  scanf("%s", class1[i].name);  
  printf("请输入年龄: ");  
  scanf("%d", &class1[i].age);  
  if (class1[i].age < 0)   
   break;  
  printf("请输入身高: ");  
  scanf("%f", &class1[i].height);  
  stu_count++;  
 }  
 for (int i = 0; i < stu_count; i++){  
  printf("name:%s, age:%d, height:%f\n",   
    class1[i].name,   
    class1[i].age,   
    class1[i].height);   
 }  
  
 return 0;  
}
```

## 指针成员访问运算符`->`

用于通过指针访问结构体成员。

**语法：**

```
结构体指针->成员变量
```


结构体参数传递：

调用函数，当使用结构传值时，结构体变量会整体复制，效率较低，因此结构体类型的变量传参时经常使用传址操作。

```c
#include <stdio.h>  
  
struct student {  
   char name[50];  // 姓名  
   int age;   // 年龄  
   float height; // 身高  
};  
  
int myadd(int x, int y) {  
 return x + y;  
}  
void print_stu_info(const struct student *stu) {  
 stu->age = 20; //报错  
 printf("name:%s, age:%d, height:%f\n", stu->name, stu->age, stu->height);  
}  
  
int main(int argc, char * argv[]) {  
 struct student zhang3 = {"zhangsan", 18, 1.80 };  
  
 print_stu_info(&zhang3);  
 // 通过结构体变量来访问成员  
 printf("name:%s, age:%d, height:%f\n", zhang3.name, zhang3.age, zhang3.height);  
  
 return 0;  
}
```

## 结构体字节对齐

尽可能让结构体的每一个成员都占据一个内存单元，提高程序运行的速度。

```c
struct student {  
    unsigned char age;  
    double x;  
    unsigned short score;  
};  
sizeof (struct student) // 24(8字节对齐）
```
