## 队列（Queue）

队列是一种特殊的线性表，只允许在表的一端进行插入操作（队尾），在另一端进行删除操作（队头），遵循**先进先出（FIFO, First In First Out）**的原则。

### 核心特性
- **操作受限**：只能在队尾插入，队头删除
- **FIFO原则**：先进队列的元素最先出队列
- **两端操作**：队头（front）用于删除，队尾（rear）用于插入

### 应用场景
- **任务调度**：操作系统中的进程调度
- **缓冲区管理**：数据流缓冲
- **广度优先搜索**：图和树的遍历算法
- **打印队列**：打印机任务管理
- **消息队列**：异步通信机制

## 队列的实现方式

### 1. 顺序队列（基于数组实现）

#### 数据结构定义
```c
#define QUEUE_SIZE 20

struct queue {
    int data[QUEUE_SIZE];  // 存储队列元素的数组
    int front;             // 队头指针
    int rear;              // 队尾指针
};
```

#### 初始化队列
```c
void queue_init(struct queue *q) {
    if (q == NULL) return;
    q->front = 0;
    q->rear = 0;
}
```

#### 判空操作
```c
int queue_is_empty(const struct queue *q) {
    if (q == NULL) return -1;
    return q->front == q->rear;
}
```

#### 判满操作（普通顺序队列）
```c
int queue_is_full(const struct queue *q) {
    if (q == NULL) return -1;
    return q->rear == QUEUE_SIZE;
}
```

#### 入队操作
```c
int queue_enqueue(struct queue *q, int value) {
    if (q == NULL) return 0;
    if (queue_is_full(q)) return 0;  // 队列满无法入队
  
    q->data[q->rear++] = value;
    return 1;
}
```

#### 出队操作
```c
int queue_dequeue(struct queue *q, int *value) {
    if (q == NULL || value == NULL) return 0;
    if (queue_is_empty(q)) return 0;  // 队列空无法出队
  
    *value = q->data[q->front++];
    return 1;
}
```

#### 获取队头元素（不删除）
```c
int queue_front(const struct queue *q, int *value) {
    if (q == NULL || value == NULL) return 0;
    if (queue_is_empty(q)) return 0;
  
    *value = q->data[q->front];
    return 1;
}
```

#### 获取队列元素个数
```c
int queue_size(const struct queue *q) {
    if (q == NULL) return -1;
    return q->rear - q->front;
}
```

#### 清空队列
```c
void queue_clear(struct queue *q) {
    if (q == NULL) return;
    q->front = 0;
    q->rear = 0;
}
```

#### 打印队列内容
```c
void queue_print(const struct queue *q) {
    if (q == NULL) return;
    if (queue_is_empty(q)) {
        printf("队列为空\n");
        return;
    }
  
    printf("队列内容（从队头到队尾）：");
    for (int i = q->front; i < q->rear; i++) {
        printf("%d ", q->data[i]);
    }
    printf("\n");
}
```

### 2. 循环队列（解决假溢出问题）

普通顺序队列出队后，前面的空间无法利用，造成"假溢出"现象。循环队列通过**模运算**使队列空间循环利用。

#### 数据结构定义
```c
#define CIRCULAR_QUEUE_SIZE 20

struct circular_queue {
    int data[CIRCULAR_QUEUE_SIZE];
    int front;      // 队头指针
    int rear;       // 队尾指针
    int size;       // 当前元素个数
};
```

#### 初始化循环队列
```c
void circular_queue_init(struct circular_queue *cq) {
    if (cq == NULL) return;
    cq->front = 0;
    cq->rear = 0;
    cq->size = 0;
}
```

#### 判空操作
```c
int circular_queue_is_empty(const struct circular_queue *cq) {
    if (cq == NULL) return -1;
    return cq->size == 0;
}
```

#### 判满操作（少用一个空间法）
```c
int circular_queue_is_full(const struct circular_queue *cq) {
    if (cq == NULL) return -1;
    return cq->size == CIRCULAR_QUEUE_SIZE - 1;
}
```

#### 入队操作
```c
int circular_queue_enqueue(struct circular_queue *cq, int value) {
    if (cq == NULL) return 0;
    if (circular_queue_is_full(cq)) return 0;
  
    cq->data[cq->rear] = value;
    cq->rear = (cq->rear + 1) % CIRCULAR_QUEUE_SIZE;
    cq->size++;
    return 1;
}
```

#### 出队操作
```c
int circular_queue_dequeue(struct circular_queue *cq, int *value) {
    if (cq == NULL || value == NULL) return 0;
    if (circular_queue_is_empty(cq)) return 0;
  
    *value = cq->data[cq->front];
    cq->front = (cq->front + 1) % CIRCULAR_QUEUE_SIZE;
    cq->size--;
    return 1;
}
```

#### 获取队头元素
```c
int circular_queue_front(const struct circular_queue *cq, int *value) {
    if (cq == NULL || value == NULL) return 0;
    if (circular_queue_is_empty(cq)) return 0;
  
    *value = cq->data[cq->front];
    return 1;
}
```

#### 获取队列大小
```c
int circular_queue_size(const struct circular_queue *cq) {
    if (cq == NULL) return -1;
    return cq->size;
}
```

#### 打印循环队列
```c
void circular_queue_print(const struct circular_queue *cq) {
    if (cq == NULL) return;
    if (circular_queue_is_empty(cq)) {
        printf("循环队列为空\n");
        return;
    }
  
    printf("循环队列内容：");
    int i = cq->front;
    int count = 0;
    while (count < cq->size) {
        printf("%d ", cq->data[i]);
        i = (i + 1) % CIRCULAR_QUEUE_SIZE;
        count++;
    }
    printf("\n");
}
```

### 3. 链式队列（基于链表实现）

#### 节点定义
```c
struct queue_node {
    int data;
    struct queue_node *next;
};

struct linked_queue {
    struct queue_node *front;  // 队头指针
    struct queue_node *rear;   // 队尾指针
    int size;                  // 队列元素个数
};
```

#### 初始化链式队列
```c
void linked_queue_init(struct linked_queue *lq) {
    if (lq == NULL) return;
    lq->front = NULL;
    lq->rear = NULL;
    lq->size = 0;
}
```

#### 判空操作
```c
int linked_queue_is_empty(const struct linked_queue *lq) {
    if (lq == NULL) return -1;
    return lq->front == NULL;
}
```

#### 入队操作
```c
int linked_queue_enqueue(struct linked_queue *lq, int value) {
    if (lq == NULL) return 0;
  
    struct queue_node *new_node = (struct queue_node*)malloc(sizeof(struct queue_node));
    if (new_node == NULL) return 0;
  
    new_node->data = value;
    new_node->next = NULL;
  
    if (linked_queue_is_empty(lq)) {
        lq->front = lq->rear = new_node;
    } else {
        lq->rear->next = new_node;
        lq->rear = new_node;
    }
  
    lq->size++;
    return 1;
}
```

#### 出队操作
```c
int linked_queue_dequeue(struct linked_queue *lq, int *value) {
    if (lq == NULL || value == NULL) return 0;
    if (linked_queue_is_empty(lq)) return 0;
  
    struct queue_node *temp = lq->front;
    *value = temp->data;
  
    lq->front = lq->front->next;
    if (lq->front == NULL) {
        lq->rear = NULL;  // 队列变空
    }
  
    free(temp);
    lq->size--;
    return 1;
}
```

#### 获取队头元素
```c
int linked_queue_front(const struct linked_queue *lq, int *value) {
    if (lq == NULL || value == NULL) return 0;
    if (linked_queue_is_empty(lq)) return 0;
  
    *value = lq->front->data;
    return 1;
}
```

#### 获取队列大小
```c
int linked_queue_size(const struct linked_queue *lq) {
    if (lq == NULL) return -1;
    return lq->size;
}
```

#### 销毁链式队列
```c
void linked_queue_destroy(struct linked_queue *lq) {
    if (lq == NULL) return;
  
    while (lq->front != NULL) {
        struct queue_node *temp = lq->front;
        lq->front = lq->front->next;
        free(temp);
    }
  
    lq->rear = NULL;
    lq->size = 0;
}
```

### 三种实现方式对比

| 特性 | 顺序队列 | 循环队列 | 链式队列 |
|------|----------|----------|----------|
| 存储方式 | 连续数组 | 连续数组 | 链表 |
| 空间效率 | 可能浪费空间 | 高效利用空间 | 按需分配 |
| 时间复杂度 | 所有操作O(1) | 所有操作O(1) | 所有操作O(1) |
| 内存开销 | 无额外开销 | 无额外开销 | 每节点额外指针 |
| 容量限制 | 固定大小 | 固定大小 | 理论无限 |
| 假溢出 | 存在 | 不存在 | 不存在 |

### 队列的应用实例

#### 1. 银行排队叫号系统
```c
struct bank_queue {
    struct circular_queue queue;
    int current_number;  // 当前号码
};

void bank_queue_init(struct bank_queue *bq) {
    circular_queue_init(&bq->queue);
    bq->current_number = 1;
}

int bank_queue_take_number(struct bank_queue *bq) {
    int number = bq->current_number++;
    if (circular_queue_enqueue(&bq->queue, number)) {
        return number;
    }
    return -1;  // 队列满
}

int bank_queue_call_number(struct bank_queue *bq) {
    int number;
    if (circular_queue_dequeue(&bq->queue, &number)) {
        return number;
    }
    return -1;  // 队列空
}
```

#### 2. 生产者-消费者模型（简化的缓冲区）
```c
struct buffer_queue {
    struct linked_queue queue;
    int max_size;
};

int buffer_enqueue(struct buffer_queue *buf, int data) {
    if (buf->queue.size >= buf->max_size) {
        return 0;  // 缓冲区满
    }
    return linked_queue_enqueue(&buf->queue, data);
}

int buffer_dequeue(struct buffer_queue *buf, int *data) {
    if (linked_queue_is_empty(&buf->queue)) {
        return 0;  // 缓冲区空
    }
    return linked_queue_dequeue(&buf->queue, data);
}
```

### 队列的性能分析

| 操作 | 时间复杂度 | 空间复杂度 |
|------|------------|------------|
| 入队 | O(1) | O(1) |
| 出队 | O(1) | O(1) |
| 获取队头 | O(1) | O(1) |
| 判空/判满 | O(1) | O(1) |

### 注意事项
1. **边界条件处理**：空队列不能出队，满队列不能入队
2. **循环队列实现**：注意模运算和判断满队列的条件
3. **内存管理**：链式队列要及时释放内存
4. **线程安全**：多线程环境下需要加锁保护
