## 双端队列（Deque, Double-ended Queue）

双端队列是一种允许在两端进行插入和删除操作的线性数据结构，结合了栈和队列的特性。

### 核心特性
- **两端操作**：可以在队首和队尾都进行插入和删除操作
- **灵活性**：兼具栈（LIFO）和队列（FIFO）的功能
- **动态性**：长度可以动态变化

### 应用场景
- **滑动窗口算法**：求解数组中滑动窗口的最大值/最小值
- **回文检测**：利用双端队列两端操作特性检测回文
- **工作调度**：支持高优先级任务插队的调度系统
- **浏览器历史记录**：前进后退功能的实现

## 双端队列的实现方式

### 1. 基于双向链表的实现

#### 节点定义
```c
struct deque_node {
    int data;
    struct deque_node *prev;  // 指向前驱节点
    struct deque_node *next;  // 指向后继节点
};

struct deque {
    struct deque_node *front;  // 队首指针
    struct deque_node *rear;   // 队尾指针
    int size;                  // 元素个数
};
```

#### 初始化双端队列
```c
void deque_init(struct deque *dq) {
    if (dq == NULL) return;
    dq->front = NULL;
    dq->rear = NULL;
    dq->size = 0;
}
```

#### 判空操作
```c
int deque_is_empty(const struct deque *dq) {
    if (dq == NULL) return -1;
    return dq->size == 0;
}
```

#### 在队首插入元素
```c
int deque_push_front(struct deque *dq, int value) {
    if (dq == NULL) return 0;
  
    struct deque_node *new_node = (struct deque_node*)malloc(sizeof(struct deque_node));
    if (new_node == NULL) return 0;
  
    new_node->data = value;
    new_node->prev = NULL;
    new_node->next = dq->front;
  
    if (dq->front != NULL) {
        dq->front->prev = new_node;
    } else {
        dq->rear = new_node;  // 如果队列为空，队尾也指向新节点
    }
  
    dq->front = new_node;
    dq->size++;
    return 1;
}
```

#### 在队尾插入元素
```c
int deque_push_back(struct deque *dq, int value) {
    if (dq == NULL) return 0;
  
    struct deque_node *new_node = (struct deque_node*)malloc(sizeof(struct deque_node));
    if (new_node == NULL) return 0;
  
    new_node->data = value;
    new_node->prev = dq->rear;
    new_node->next = NULL;
  
    if (dq->rear != NULL) {
        dq->rear->next = new_node;
    } else {
        dq->front = new_node;  // 如果队列为空，队首也指向新节点
    }
  
    dq->rear = new_node;
    dq->size++;
    return 1;
}
```

#### 从队首删除元素
```c
int deque_pop_front(struct deque *dq, int *value) {
    if (dq == NULL || value == NULL) return 0;
    if (deque_is_empty(dq)) return 0;
  
    struct deque_node *temp = dq->front;
    *value = temp->data;
  
    dq->front = dq->front->next;
    if (dq->front != NULL) {
        dq->front->prev = NULL;
    } else {
        dq->rear = NULL;  // 队列变空
    }
  
    free(temp);
    dq->size--;
    return 1;
}
```

#### 从队尾删除元素
```c
int deque_pop_back(struct deque *dq, int *value) {
    if (dq == NULL || value == NULL) return 0;
    if (deque_is_empty(dq)) return 0;
  
    struct deque_node *temp = dq->rear;
    *value = temp->data;
  
    dq->rear = dq->rear->prev;
    if (dq->rear != NULL) {
        dq->rear->next = NULL;
    } else {
        dq->front = NULL;  // 队列变空
    }
  
    free(temp);
    dq->size--;
    return 1;
}
```

#### 获取队首元素（不删除）
```c
int deque_front(const struct deque *dq, int *value) {
    if (dq == NULL || value == NULL) return 0;
    if (deque_is_empty(dq)) return 0;
  
    *value = dq->front->data;
    return 1;
}
```

#### 获取队尾元素（不删除）
```c
int deque_back(const struct deque *dq, int *value) {
    if (dq == NULL || value == NULL) return 0;
    if (deque_is_empty(dq)) return 0;
  
    *value = dq->rear->data;
    return 1;
}
```

#### 获取双端队列大小
```c
int deque_size(const struct deque *dq) {
    if (dq == NULL) return -1;
    return dq->size;
}
```

#### 清空双端队列
```c
void deque_clear(struct deque *dq) {
    if (dq == NULL) return;
  
    while (dq->front != NULL) {
        struct deque_node *temp = dq->front;
        dq->front = dq->front->next;
        free(temp);
    }
  
    dq->rear = NULL;
    dq->size = 0;
}
```

#### 打印双端队列（从队首到队尾）
```c
void deque_print_forward(const struct deque *dq) {
    if (dq == NULL) return;
    if (deque_is_empty(dq)) {
        printf("双端队列为空\n");
        return;
    }
  
    printf("双端队列内容（从队首到队尾）：");
    struct deque_node *current = dq->front;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}
```

#### 打印双端队列（从队尾到队首）
```c
void deque_print_backward(const struct deque *dq) {
    if (dq == NULL) return;
    if (deque_is_empty(dq)) {
        printf("双端队列为空\n");
        return;
    }
  
    printf("双端队列内容（从队尾到队首）：");
    struct deque_node *current = dq->rear;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->prev;
    }
    printf("\n");
}
```

### 2. 基于循环数组的实现

#### 数据结构定义
```c
#define DEQUE_CAPACITY 20

struct array_deque {
    int data[DEQUE_CAPACITY];
    int front;      // 队首索引
    int rear;       // 队尾索引
    int size;       // 当前元素个数
};
```

#### 初始化数组双端队列
```c
void array_deque_init(struct array_deque *adq) {
    if (adq == NULL) return;
    adq->front = 0;
    adq->rear = -1;
    adq->size = 0;
}
```

#### 判空操作
```c
int array_deque_is_empty(const struct array_deque *adq) {
    if (adq == NULL) return -1;
    return adq->size == 0;
}
```

#### 判满操作
```c
int array_deque_is_full(const struct array_deque *adq) {
    if (adq == NULL) return -1;
    return adq->size == DEQUE_CAPACITY;
}
```

#### 在队首插入元素
```c
int array_deque_push_front(struct array_deque *adq, int value) {
    if (adq == NULL) return 0;
    if (array_deque_is_full(adq)) return 0;
  
    adq->front = (adq->front - 1 + DEQUE_CAPACITY) % DEQUE_CAPACITY;
    adq->data[adq->front] = value;
    adq->size++;
    return 1;
}
```

#### 在队尾插入元素
```c
int array_deque_push_back(struct array_deque *adq, int value) {
    if (adq == NULL) return 0;
    if (array_deque_is_full(adq)) return 0;
  
    adq->rear = (adq->rear + 1) % DEQUE_CAPACITY;
    adq->data[adq->rear] = value;
    adq->size++;
    return 1;
}
```

#### 从队首删除元素
```c
int array_deque_pop_front(struct array_deque *adq, int *value) {
    if (adq == NULL || value == NULL) return 0;
    if (array_deque_is_empty(adq)) return 0;
  
    *value = adq->data[adq->front];
    adq->front = (adq->front + 1) % DEQUE_CAPACITY;
    adq->size--;
    return 1;
}
```

#### 从队尾删除元素
```c
int array_deque_pop_back(struct array_deque *adq, int *value) {
    if (adq == NULL || value == NULL) return 0;
    if (array_deque_is_empty(adq)) return 0;
  
    *value = adq->data[adq->rear];
    adq->rear = (adq->rear - 1 + DEQUE_CAPACITY) % DEQUE_CAPACITY;
    adq->size--;
    return 1;
}
```

#### 获取队首和队尾元素
```c
int array_deque_front(const struct array_deque *adq, int *value) {
    if (adq == NULL || value == NULL) return 0;
    if (array_deque_is_empty(adq)) return 0;
  
    *value = adq->data[adq->front];
    return 1;
}

int array_deque_back(const struct array_deque *adq, int *value) {
    if (adq == NULL || value == NULL) return 0;
    if (array_deque_is_empty(adq)) return 0;
  
    *value = adq->data[adq->rear];
    return 1;
}
```

#### 获取尺寸
```c
int array_deque_size(const struct array_deque *adq) {
    if (adq == NULL) return -1;
    return adq->size;
}
```

## 双端队列的应用实例

### 1. 滑动窗口最大值问题
给定一个数组和一个窗口大小k，找出每个窗口中的最大值。

```c
// 使用双端队列维护窗口中元素的索引
void sliding_window_maximum(const int *arr, int n, int k, int *result) {
    struct deque dq;
    deque_init(&dq);
  
    // 处理前k个元素
    for (int i = 0; i < k; i++) {
        // 移除所有小于当前元素的元素（从队尾）
        while (!deque_is_empty(&dq)) {
            int back_val;
            deque_back(&dq, &back_val);
            if (arr[back_val] < arr[i]) {
                int temp;
                deque_pop_back(&dq, &temp);
            } else {
                break;
            }
        }
        deque_push_back(&dq, i);
    }
  
    result[0] = arr[dq.front->data];
  
    // 处理剩余元素
    for (int i = k; i < n; i++) {
        // 移除超出窗口的元素
        while (!deque_is_empty(&dq)) {
            int front_val;
            deque_front(&dq, &front_val);
            if (front_val <= i - k) {
                int temp;
                deque_pop_front(&dq, &temp);
            } else {
                break;
            }
        }
      
        // 移除所有小于当前元素的元素（从队尾）
        while (!deque_is_empty(&dq)) {
            int back_val;
            deque_back(&dq, &back_val);
            if (arr[back_val] < arr[i]) {
                int temp;
                deque_pop_back(&dq, &temp);
            } else {
                break;
            }
        }
      
        deque_push_back(&dq, i);
        int front_index;
        deque_front(&dq, &front_index);
        result[i - k + 1] = arr[front_index];
    }
  
    deque_clear(&dq);
}
```

### 2. 回文检测
```c
int is_palindrome(const char *str) {
    struct deque dq;
    deque_init(&dq);
  
    // 将所有字符加入双端队列
    for (int i = 0; str[i] != '\0'; i++) {
        if (isalnum(str[i])) {  // 只考虑字母和数字
            deque_push_back(&dq, tolower(str[i]));
        }
    }
  
    // 从两端比较字符
    while (deque_size(&dq) > 1) {
        int front_char, back_char;
        deque_pop_front(&dq, &front_char);
        deque_pop_back(&dq, &back_char);
      
        if (front_char != back_char) {
            deque_clear(&dq);
            return 0;
        }
    }
  
    deque_clear(&dq);
    return 1;
}
```

### 3. 工作任务调度
```c
struct task_scheduler {
    struct deque high_priority_tasks;  // 高优先级任务队列
    struct deque normal_tasks;         // 普通任务队列
};

void task_scheduler_init(struct task_scheduler *scheduler) {
    deque_init(&scheduler->high_priority_tasks);
    deque_init(&scheduler->normal_tasks);
}

// 添加高优先级任务（插入到队首）
void add_high_priority_task(struct task_scheduler *scheduler, int task_id) {
    deque_push_front(&scheduler->high_priority_tasks, task_id);
}

// 添加普通任务（添加到队尾）
void add_normal_task(struct task_scheduler *scheduler, int task_id) {
    deque_push_back(&scheduler->normal_tasks, task_id);
}

// 获取下一个要处理的任务
int get_next_task(struct task_scheduler *scheduler) {
    int task_id;
  
    // 优先处理高优先级任务
    if (!deque_is_empty(&scheduler->high_priority_tasks)) {
        deque_pop_front(&scheduler->high_priority_tasks, &task_id);
        return task_id;
    }
  
    // 处理普通任务
    if (!deque_is_empty(&scheduler->normal_tasks)) {
        deque_pop_front(&scheduler->normal_tasks, &task_id);
        return task_id;
    }
  
    return -1;  // 没有任务
}
```

## 性能分析

### 时间复杂度
| 操作 | 时间复杂度 |
|------|------------|
| 在队首插入/删除 | O(1) |
| 在队尾插入/删除 | O(1) |
| 访问队首元素 | O(1) |
| 访问队尾元素 | O(1) |

### 空间复杂度
- **链式实现**：O(n)，其中n为元素个数
- **数组实现**：O(固定容量)

---

## 两种实现方式对比

| 特性 | 链式双端队列 | 数组双端队列 |
|------|--------------|--------------|
| 内存分配 | 动态分配，按需增长 | 预分配固定大小 |
| 空间效率 | 每节点额外指针开销 | 可能浪费部分空间 |
| 容量限制 | 理论上无限 | 受数组大小限制 |
| 实现复杂度 | 相对复杂 | 相对简单 |
| 内存局部性 | 较差 | 较好 |

### 注意事项
1. **内存管理**：链式实现要防止内存泄漏
2. **循环数组边界**：数组实现要注意模运算和边界条件
3. **错误处理**：各种异常情况的处理
4. **应用选择**：根据具体应用场景选择合适的实现方式
