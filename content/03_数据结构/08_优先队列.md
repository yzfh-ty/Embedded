## 优先队列（Priority Queue）

优先队列是一种特殊的队列，其中的每个元素都有一个优先级，出队时总是优先级最高的元素先出队，而不像普通队列那样按照入队顺序出队。

### 核心特性
- **优先级驱动**：元素按照优先级排序，高优先级元素优先出队
- **动态排序**：插入新元素时自动维护优先级顺序
- **高效访问**：能够快速获取最高优先级元素

### 应用场景
- **任务调度**：操作系统进程调度、网络数据包调度
- **事件处理**：模拟器中的事件调度
- **图算法**：Dijkstra最短路径算法、Prim最小生成树算法
- **堆排序**：基于堆的排序算法
- **Huffman编码**：数据压缩算法

## 优先队列的实现方式

### 1. 基于二叉堆的实现（最大堆/最小堆）

#### 数据结构定义
```c
#define PRIORITY_QUEUE_SIZE 100

struct priority_queue {
    int data[PRIORITY_QUEUE_SIZE];  // 存储堆元素的数组
    int size;                       // 当前元素个数
    int is_max_heap;               // 1表示最大堆，0表示最小堆
};
```

#### 初始化优先队列
```c
void priority_queue_init(struct priority_queue *pq, int is_max_heap) {
    if (pq == NULL) return;
    pq->size = 0;
    pq->is_max_heap = is_max_heap;
}
```

#### 获取父节点、左右子节点索引
```c
static int parent(int i) {
    return (i - 1) / 2;
}

static int left_child(int i) {
    return 2 * i + 1;
}

static int right_child(int i) {
    return 2 * i + 2;
}
```

#### 比较元素（根据堆类型）
```c
static int compare_priority(const struct priority_queue *pq, int a, int b) {
    if (pq->is_max_heap) {
        return a > b;  // 最大堆：大的优先级高
    } else {
        return a < b;  // 最小堆：小的优先级高
    }
}
```

#### 向上调整（上浮）
```c
static void heapify_up(struct priority_queue *pq, int index) {
    while (index > 0 && compare_priority(pq, pq->data[index], pq->data[parent(index)])) {
        // 交换当前节点和父节点
        int temp = pq->data[index];
        pq->data[index] = pq->data[parent(index)];
        pq->data[parent(index)] = temp;
      
        index = parent(index);
    }
}
```

#### 向下调整（下沉）
```c
static void heapify_down(struct priority_queue *pq, int index) {
    int largest = index;
    int left = left_child(index);
    int right = right_child(index);
  
    // 找到当前节点和子节点中优先级最高的
    if (left < pq->size && compare_priority(pq, pq->data[left], pq->data[largest])) {
        largest = left;
    }
  
    if (right < pq->size && compare_priority(pq, pq->data[right], pq->data[largest])) {
        largest = right;
    }
  
    // 如果最大值不是当前节点，则交换并继续向下调整
    if (largest != index) {
        int temp = pq->data[index];
        pq->data[index] = pq->data[largest];
        pq->data[largest] = temp;
      
        heapify_down(pq, largest);
    }
}
```

#### 插入元素（入队）
```c
int priority_queue_enqueue(struct priority_queue *pq, int value) {
    if (pq == NULL || pq->size >= PRIORITY_QUEUE_SIZE) return 0;
  
    // 将新元素插入到数组末尾
    pq->data[pq->size] = value;
    pq->size++;
  
    // 上浮调整
    heapify_up(pq, pq->size - 1);
    return 1;
}
```

#### 删除最高优先级元素（出队）
```c
int priority_queue_dequeue(struct priority_queue *pq, int *value) {
    if (pq == NULL || value == NULL || pq->size <= 0) return 0;
  
    // 保存堆顶元素
    *value = pq->data[0];
  
    // 将最后一个元素移到堆顶
    pq->data[0] = pq->data[pq->size - 1];
    pq->size--;
  
    // 下沉调整
    if (pq->size > 0) {
        heapify_down(pq, 0);
    }
  
    return 1;
}
```

#### 获取最高优先级元素（不删除）
```c
int priority_queue_top(const struct priority_queue *pq, int *value) {
    if (pq == NULL || value == NULL || pq->size <= 0) return 0;
  
    *value = pq->data[0];
    return 1;
}
```

#### 判空操作
```c
int priority_queue_is_empty(const struct priority_queue *pq) {
    if (pq == NULL) return -1;
    return pq->size == 0;
}
```

#### 获取队列大小
```c
int priority_queue_size(const struct priority_queue *pq) {
    if (pq == NULL) return -1;
    return pq->size;
}
```

#### 清空队列
```c
void priority_queue_clear(struct priority_queue *pq) {
    if (pq == NULL) return;
    pq->size = 0;
}
```

#### 打印堆内容（层序遍历）
```c
void priority_queue_print(const struct priority_queue *pq) {
    if (pq == NULL) return;
    if (priority_queue_is_empty(pq)) {
        printf("优先队列为空\n");
        return;
    }
  
    printf("优先队列内容：");
    for (int i = 0; i < pq->size; i++) {
        printf("%d ", pq->data[i]);
    }
    printf("\n");
}
```

### 2. 基于链表的有序实现

#### 节点定义
```c
struct pq_node {
    int data;
    int priority;           // 优先级值，数值越大优先级越高
    struct pq_node *next;
};

struct linked_priority_queue {
    struct pq_node *head;
    int size;
};
```

#### 初始化链式优先队列
```c
void linked_pq_init(struct linked_priority_queue *lpq) {
    if (lpq == NULL) return;
    lpq->head = NULL;
    lpq->size = 0;
}
```

#### 插入元素（保持有序）
```c
int linked_pq_enqueue(struct linked_priority_queue *lpq, int value, int priority) {
    if (lpq == NULL) return 0;
  
    struct pq_node *new_node = (struct pq_node*)malloc(sizeof(struct pq_node));
    if (new_node == NULL) return 0;
  
    new_node->data = value;
    new_node->priority = priority;
  
    // 如果队列为空或新元素优先级最高
    if (lpq->head == NULL || priority > lpq->head->priority) {
        new_node->next = lpq->head;
        lpq->head = new_node;
    } else {
        // 找到合适的插入位置
        struct pq_node *current = lpq->head;
        while (current->next != NULL && current->next->priority >= priority) {
            current = current->next;
        }
        new_node->next = current->next;
        current->next = new_node;
    }
  
    lpq->size++;
    return 1;
}
```

#### 删除最高优先级元素
```c
int linked_pq_dequeue(struct linked_priority_queue *lpq, int *value) {
    if (lpq == NULL || value == NULL || lpq->head == NULL) return 0;
  
    struct pq_node *temp = lpq->head;
    *value = temp->data;
    lpq->head = lpq->head->next;
    free(temp);
    lpq->size--;
    return 1;
}
```

#### 获取最高优先级元素
```c
int linked_pq_top(const struct linked_priority_queue *lpq, int *value) {
    if (lpq == NULL || value == NULL || lpq->head == NULL) return 0;
  
    *value = lpq->head->data;
    return 1;
}
```

#### 销毁链式优先队列
```c
void linked_pq_destroy(struct linked_priority_queue *lpq) {
    if (lpq == NULL) return;
  
    while (lpq->head != NULL) {
        struct pq_node *temp = lpq->head;
        lpq->head = lpq->head->next;
        free(temp);
    }
    lpq->size = 0;
}
```

## 二叉堆的详细操作

### 堆的构建过程示例

以最大堆为例，插入元素：[4, 1, 3, 2, 16, 9, 10, 14, 8, 7]

```
初始状态: []
插入 4:  [4]
插入 1:  [4, 1]
插入 3:  [4, 1, 3]
插入 2:  [4, 2, 3, 1]
插入 16: [16, 4, 3, 1, 2]  // 16上浮到根
插入 9:  [16, 4, 9, 1, 2, 3]
插入 10: [16, 4, 10, 1, 2, 3, 9]
插入 14: [16, 14, 10, 4, 2, 3, 9, 1]
插入 8:  [16, 14, 10, 8, 2, 3, 9, 1, 4]
插入 7:  [16, 14, 10, 8, 7, 3, 9, 1, 4, 2]
```

首次出队操作（删除16）：
```
步骤1: 将最后一个元素2移动到根 [2, 14, 10, 8, 7, 3, 9, 1, 4]
步骤2: 2下沉，与14交换 [14, 2, 10, 8, 7, 3, 9, 1, 4]
步骤3: 2继续下沉，与8交换 [14, 8, 10, 2, 7, 3, 9, 1, 4]
最终堆: [14, 8, 10, 2, 7, 3, 9, 1, 4]
```

## 优先队列的高级应用

### 1. Dijkstra最短路径算法
```c
struct dijkstra_node {
    int vertex;
    int distance;
};

struct graph {
    int vertices;
    struct adjacency_list *adj_list;
};

int dijkstra_shortest_path(struct graph *g, int source, int *distances) {
    struct priority_queue pq;
    priority_queue_init(&pq, 0);  // 最小堆
  
    // 初始化距离数组
    for (int i = 0; i < g->vertices; i++) {
        distances[i] = INT_MAX;
    }
    distances[source] = 0;
  
    // 将源点加入优先队列
    priority_queue_enqueue(&pq, source * g->vertices + 0);
  
    while (!priority_queue_is_empty(&pq)) {
        int combined;
        priority_queue_dequeue(&pq, &combined);
        int u = combined / g->vertices;
        int dist = combined % g->vertices;
      
        if (dist > distances[u]) continue;
      
        // 遍历u的所有邻接点
        struct adjacency_node *adj = g->adj_list[u].head;
        while (adj != NULL) {
            int v = adj->vertex;
            int weight = adj->weight;
          
            if (distances[u] + weight < distances[v]) {
                distances[v] = distances[u] + weight;
                priority_queue_enqueue(&pq, v * g->vertices + distances[v]);
            }
            adj = adj->next;
        }
    }
  
    return 1;
}
```

### 2. 合并K个有序链表
```c
struct list_node {
    int val;
    struct list_node *next;
};

struct list_node* merge_k_sorted_lists(struct list_node **lists, int k) {
    struct priority_queue pq;
    priority_queue_init(&pq, 0);  // 最小堆
  
    // 将每个链表的头节点加入优先队列
    for (int i = 0; i < k; i++) {
        if (lists[i] != NULL) {
            priority_queue_enqueue(&pq, lists[i]->val * 1000 + i);
        }
    }
  
    struct list_node dummy;
    struct list_node *current = &dummy;
  
    while (!priority_queue_is_empty(&pq)) {
        int combined;
        priority_queue_dequeue(&pq, &combined);
        int val = combined / 1000;
        int list_index = combined % 1000;
      
        // 创建新节点
        current->next = (struct list_node*)malloc(sizeof(struct list_node));
        current->next->val = val;
        current->next->next = NULL;
        current = current->next;
      
        // 移动到下一个节点
        lists[list_index] = lists[list_index]->next;
        if (lists[list_index] != NULL) {
            priority_queue_enqueue(&pq, lists[list_index]->val * 1000 + list_index);
        }
    }
  
    return dummy.next;
}
```

### 3. 数据流的中位数
```c
struct median_finder {
    struct priority_queue max_heap;  // 存储较小的一半
    struct priority_queue min_heap;  // 存储较大的一半
};

void median_finder_init(struct median_finder *mf) {
    priority_queue_init(&mf->max_heap, 1);  // 最大堆
    priority_queue_init(&mf->min_heap, 0);  // 最小堆
}

void median_finder_add_number(struct median_finder *mf, int num) {
    // 先插入到最大堆
    priority_queue_enqueue(&mf->max_heap, num);
  
    // 将最大堆的最大值移到最小堆
    int max_val;
    priority_queue_dequeue(&mf->max_heap, &max_val);
    priority_queue_enqueue(&mf->min_heap, max_val);
  
    // 平衡两个堆的大小
    if (priority_queue_size(&mf->min_heap) > priority_queue_size(&mf->max_heap)) {
        int min_val;
        priority_queue_dequeue(&mf->min_heap, &min_val);
        priority_queue_enqueue(&mf->max_heap, min_val);
    }
}

double median_finder_find_median(struct median_finder *mf) {
    if (priority_queue_size(&mf->max_heap) > priority_queue_size(&mf->min_heap)) {
        int top;
        priority_queue_top(&mf->max_heap, &top);
        return (double)top;
    } else {
        int max_top, min_top;
        priority_queue_top(&mf->max_heap, &max_top);
        priority_queue_top(&mf->min_heap, &min_top);
        return ((double)max_top + (double)min_top) / 2.0;
    }
}
```

## 性能分析

### 时间复杂度
| 操作 | 二叉堆实现 | 有序链表实现 |
|------|------------|--------------|
| 插入元素 | O(log n) | O(n) |
| 删除最高优先级元素 | O(log n) | O(1) |
| 获取最高优先级元素 | O(1) | O(1) |
| 构建n个元素的堆 | O(n log n) | O(n²) |

### 空间复杂度
- **二叉堆**：O(n)
- **有序链表**：O(n)

## 两种实现方式对比

| 特性 | 二叉堆实现 | 有序链表实现 |
|------|------------|--------------|
| 插入效率 | O(log n) | O(n) |
| 删除效率 | O(log n) | O(1) |
| 内存使用 | 紧凑的数组 | 每节点额外指针 |
| 缓存友好性 | 较好 | 一般 |
| 实现复杂度 | 中等 | 简单 |
| 随机访问 | O(n) | O(n) |

### 注意事项
1. **堆性质维护**：插入和删除时必须保持堆的性质
2. **索引计算**：数组实现中要注意父子节点的索引关系
3. **内存管理**：链式实现要避免内存泄漏
4. **优先级定义**：明确优先级的比较规则
5. **并发安全**：多线程环境下需要同步机制
