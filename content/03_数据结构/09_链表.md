# 链表（Linked List）

链表是一种线性数据结构，由一系列节点组成，每个节点包含数据域和指向下一个节点的指针。与数组不同，链表的元素在内存中不是连续存储的。

## 核心特性
- **动态大小**：可以根据需要动态增长或缩小
- **非连续存储**：节点在内存中的位置可以是任意的
- **高效的插入/删除**：不需要移动其他元素
- **额外内存开销**：每个节点都需要存储指针

## 应用场景
- **动态内存分配**：程序运行时需要动态管理内存
- **多项式运算**：表示和操作多项式
- **图和树的实现**：作为基础数据结构
- **LRU缓存**：最近最少使用缓存算法
- **浏览器历史记录**：前进后退功能
- **音乐播放列表**：上一首/下一首功能

## 单向链表（Singly Linked List）

### 节点定义
```c
struct list_node {
    int data;                    // 数据域
    struct list_node *next;      // 指向下一个节点的指针
};

struct linked_list {
    struct list_node *head;      // 指向链表头节点的指针
    int size;                    // 链表中节点的个数
};
```

### 初始化链表
```c
void linked_list_init(struct linked_list *list) {
    if (list == NULL) return;
    list->head = NULL;
    list->size = 0;
}
```

### 创建新节点
```c
static struct list_node* create_node(int data) {
    struct list_node *new_node = (struct list_node*)malloc(sizeof(struct list_node));
    if (new_node == NULL) return NULL;
  
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}
```

### 在链表头部插入节点
```c
int linked_list_push_front(struct linked_list *list, int data) {
    if (list == NULL) return 0;
  
    struct list_node *new_node = create_node(data);
    if (new_node == NULL) return 0;
  
    new_node->next = list->head;
    list->head = new_node;
    list->size++;
    return 1;
}
```

### 在链表尾部插入节点
```c
int linked_list_push_back(struct linked_list *list, int data) {
    if (list == NULL) return 0;
  
    struct list_node *new_node = create_node(data);
    if (new_node == NULL) return 0;
  
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        struct list_node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
  
    list->size++;
    return 1;
}
```

### 在指定位置插入节点
```c
int linked_list_insert(struct linked_list *list, int index, int data) {
    if (list == NULL || index < 0 || index > list->size) return 0;
  
    if (index == 0) {
        return linked_list_push_front(list, data);
    }
  
    struct list_node *new_node = create_node(data);
    if (new_node == NULL) return 0;
  
    struct list_node *current = list->head;
    for (int i = 0; i < index - 1; i++) {
        current = current->next;
    }
  
    new_node->next = current->next;
    current->next = new_node;
    list->size++;
    return 1;
}
```

### 删除链表头部节点
```c
int linked_list_pop_front(struct linked_list *list, int *data) {
    if (list == NULL || data == NULL || list->head == NULL) return 0;
  
    struct list_node *temp = list->head;
    *data = temp->data;
    list->head = list->head->next;
    free(temp);
    list->size--;
    return 1;
}
```

### 删除链表尾部节点
```c
int linked_list_pop_back(struct linked_list *list, int *data) {
    if (list == NULL || data == NULL || list->head == NULL) return 0;
  
    if (list->head->next == NULL) {
        // 只有一个节点
        *data = list->head->data;
        free(list->head);
        list->head = NULL;
    } else {
        // 找到倒数第二个节点
        struct list_node *current = list->head;
        while (current->next->next != NULL) {
            current = current->next;
        }
        *data = current->next->data;
        free(current->next);
        current->next = NULL;
    }
  
    list->size--;
    return 1;
}
```

### 删除指定位置的节点
```c
int linked_list_remove(struct linked_list *list, int index, int *data) {
    if (list == NULL || data == NULL || index < 0 || index >= list->size) return 0;
  
    if (index == 0) {
        return linked_list_pop_front(list, data);
    }
  
    struct list_node *current = list->head;
    for (int i = 0; i < index - 1; i++) {
        current = current->next;
    }
  
    struct list_node *to_remove = current->next;
    *data = to_remove->data;
    current->next = to_remove->next;
    free(to_remove);
    list->size--;
    return 1;
}
```

### 删除第一个匹配的节点
```c
int linked_list_remove_value(struct linked_list *list, int value) {
    if (list == NULL || list->head == NULL) return 0;
  
    if (list->head->data == value) {
        struct list_node *temp = list->head;
        list->head = list->head->next;
        free(temp);
        list->size--;
        return 1;
    }
  
    struct list_node *current = list->head;
    while (current->next != NULL && current->next->data != value) {
        current = current->next;
    }
  
    if (current->next != NULL) {
        struct list_node *to_remove = current->next;
        current->next = to_remove->next;
        free(to_remove);
        list->size--;
        return 1;
    }
  
    return 0;  // 未找到匹配的节点
}
```

### 查找节点
```c
int linked_list_find(const struct linked_list *list, int data) {
    if (list == NULL) return -1;
  
    struct list_node *current = list->head;
    int index = 0;
  
    while (current != NULL) {
        if (current->data == data) {
            return index;
        }
        current = current->next;
        index++;
    }
  
    return -1;  // 未找到
}
```

### 修改指定位置的节点数据
```c
int linked_list_set(struct linked_list *list, int index, int data) {
```

## 双向链表（Doubly Linked List）

### 节点定义
```c
struct dlist_node {
    int data;
    struct dlist_node *prev;    // 指向前驱节点
    struct dlist_node *next;    // 指向后继节点
};

struct doubly_linked_list {
    struct dlist_node *head;    // 头节点指针
    struct dlist_node *tail;    // 尾节点指针
    int size;
};
```

### 初始化双向链表
```c
void doubly_linked_list_init(struct doubly_linked_list *dlist) {
    if (dlist == NULL) return;
    dlist->head = NULL;
    dlist->tail = NULL;
    dlist->size = 0;
}
```

### 创建新节点
```c
static struct dlist_node* create_dnode(int data) {
    struct dlist_node *new_node = (struct dlist_node*)malloc(sizeof(struct dlist_node));
    if (new_node == NULL) return NULL;
  
    new_node->data = data;
    new_node->prev = NULL;
    new_node->next = NULL;
    return new_node;
}
```

### 在头部插入节点
```c
int doubly_linked_list_push_front(struct doubly_linked_list *dlist, int data) {
    if (dlist == NULL) return 0;
  
    struct dlist_node *new_node = create_dnode(data);
    if (new_node == NULL) return 0;
  
    if (dlist->head == NULL) {
        dlist->head = dlist->tail = new_node;
    } else {
        new_node->next = dlist->head;
        dlist->head->prev = new_node;
        dlist->head = new_node;
    }
  
    dlist->size++;
    return 1;
}
```

### 在尾部插入节点
```c
int doubly_linked_list_push_back(struct doubly_linked_list *dlist, int data) {
    if (dlist == NULL) return 0;
  
    struct dlist_node *new_node = create_dnode(data);
    if (new_node == NULL) return 0;
  
    if (dlist->tail == NULL) {
        dlist->head = dlist->tail = new_node;
    } else {
        new_node->prev = dlist->tail;
        dlist->tail->next = new_node;
        dlist->tail = new_node;
    }
  
    dlist->size++;
    return 1;
}
```

### 从头部删除节点
```c
int doubly_linked_list_pop_front(struct doubly_linked_list *dlist, int *data) {
    if (dlist == NULL || data == NULL || dlist->head == NULL) return 0;
  
    struct dlist_node *temp = dlist->head;
    *data = temp->data;
  
    if (dlist->head == dlist->tail) {
        // 只有一个节点
        dlist->head = dlist->tail = NULL;
    } else {
        dlist->head = dlist->head->next;
        dlist->head->prev = NULL;
    }
  
    free(temp);
    dlist->size--;
    return 1;
}
```

### 从尾部删除节点
```c
int doubly_linked_list_pop_back(struct doubly_linked_list *dlist, int *data) {
    if (dlist == NULL || data == NULL || dlist->tail == NULL) return 0;
  
    struct dlist_node *temp = dlist->tail;
    *data = temp->data;
  
    if (dlist->head == dlist->tail) {
        // 只有一个节点
        dlist->head = dlist->tail = NULL;
    } else {
        dlist->tail = dlist->tail->prev;
        dlist->tail->next = NULL;
    }
  
    free(temp);
    dlist->size--;
    return 1;
}
```

### 在指定位置插入节点
```c
int doubly_linked_list_insert(struct doubly_linked_list *dlist, int index, int data) {
    if (dlist == NULL || index < 0 || index > dlist->size) return 0;
  
    if (index == 0) {
        return doubly_linked_list_push_front(dlist, data);
    }
  
    if (index == dlist->size) {
        return doubly_linked_list_push_back(dlist, data);
    }
  
    struct dlist_node *new_node = create_dnode(data);
    if (new_node == NULL) return 0;
  
    struct dlist_node *current;
    if (index < dlist->size / 2) {
        // 从头部开始查找
        current = dlist->head;
        for (int i = 0; i < index; i++) {
            current = current->next;
        }
    } else {
        // 从尾部开始查找
        current = dlist->tail;
        for (int i = dlist->size - 1; i > index; i--) {
            current = current->prev;
        }
    }
  
    new_node->next = current;
    new_node->prev = current->prev;
    current->prev->next = new_node;
    current->prev = new_node;
  
    dlist->size++;
    return 1;
}
```

### 删除指定位置的节点
```c
int doubly_linked_list_remove(struct doubly_linked_list *dlist, int index, int *data) {
    if (dlist == NULL || data == NULL || index < 0 || index >= dlist->size) return 0;
  
    if (index == 0) {
        return doubly_linked_list_pop_front(dlist, data);
    }
  
    if (index == dlist->size - 1) {
        return doubly_linked_list_pop_back(dlist, data);
    }
  
    struct dlist_node *current;
    if (index < dlist->size / 2) {
        // 从头部开始查找
        current = dlist->head;
        for (int i = 0; i < index; i++) {
            current = current->next;
        }
    } else {
        // 从尾部开始查找
        current = dlist->tail;
        for (int i = dlist->size - 1; i > index; i--) {
            current = current->prev;
        }
    }
  
    *data = current->data;
    current->prev->next = current->next;
    current->next->prev = current->prev;
    free(current);
    dlist->size--;
    return 1;
}
```

### 清空双向链表
```c
void doubly_linked_list_clear(struct doubly_linked_list *dlist) {
    if (dlist == NULL) return;
  
    while (dlist->head != NULL) {
        struct dlist_node *temp = dlist->head;
        dlist->head = dlist->head->next;
        free(temp);
    }
  
    dlist->tail = NULL;
    dlist->size = 0;
}
```

## 循环链表（Circular Linked List）

### 单向循环链表
```c
struct circular_list_node {
    int data;
    struct circular_list_node *next;
};

struct circular_linked_list {
    struct circular_list_node *head;
    int size;
};
```

### 初始化循环链表
```c
void circular_linked_list_init(struct circular_linked_list *clist) {
    if (clist == NULL) return;
    clist->head = NULL;
    clist->size = 0;
}
```

### 在循环链表中插入节点
```c
int circular_linked_list_insert(struct circular_linked_list *clist, int data) {
    if (clist == NULL) return 0;
  
    struct circular_list_node *new_node = (struct circular_list_node*)malloc(sizeof(struct circular_list_node));
    if (new_node == NULL) return 0;
  
    new_node->data = data;
  
    if (clist->head == NULL) {
        // 空链表
        clist->head = new_node;
        new_node->next = new_node;  // 指向自己
    } else {
        // 插入到头部
        struct circular_list_node *last = clist->head;
        while (last->next != clist->head) {
            last = last->next;
        }
        new_node->next = clist->head;
        last->next = new_node;
        clist->head = new_node;
    }
  
    clist->size++;
    return 1;
}
```

### 删除循环链表中的节点
```c
int circular_linked_list_remove(struct circular_linked_list *clist, int data) {
    if (clist == NULL || clist->head == NULL) return 0;
  
    struct circular_list_node *current = clist->head;
    struct circular_list_node *prev = NULL;
  
    // 找到最后一个节点
    struct circular_list_node *last = clist->head;
    while (last->next != clist->head) {
        last = last->next;
    }
  
    do {
        if (current->data == data) {
            if (current == clist->head) {
                // 删除头节点
                if (clist->head->next == clist->head) {
                    // 只有一个节点
                    free(clist->head);
                    clist->head = NULL;
                } else {
                    last->next = clist->head->next;
                    struct circular_list_node *temp = clist->head;
                    clist->head = clist->head->next;
                    free(temp);
                }
            } else {
                // 删除中间或尾部节点
                prev->next = current->next;
                free(current);
            }
            clist->size--;
            return 1;
        }
        prev = current;
        current = current->next;
    } while (current != clist->head);
  
    return 0;  // 未找到要删除的节点
}
```

## 链表的经典应用

### 1. 链表反转
```c
// 单向链表反转
void linked_list_reverse(struct linked_list *list) {
    if (list == NULL || list->head == NULL) return;
  
    struct list_node *prev = NULL;
    struct list_node *current = list->head;
    struct list_node *next = NULL;
  
    while (current != NULL) {
        next = current->next;
        current->next = prev;
        prev = current;
        current = next;
    }
  
    list->head = prev;
}

// 双向链表反转
void doubly_linked_list_reverse(struct doubly_linked_list *dlist) {
    if (dlist == NULL || dlist->head == NULL) return;
  
    struct dlist_node *current = dlist->head;
    struct dlist_node *temp = NULL;
  
    while (current != NULL) {
        temp = current->prev;
        current->prev = current->next;
        current->next = temp;
        current = current->prev;
    }
  
    if (temp != NULL) {
        dlist->head = temp->prev;
    }
}
```

### 2. 检测链表是否有环
```c
// Floyd判圈算法（快慢指针法）
int linked_list_has_cycle(const struct linked_list *list) {
    if (list == NULL || list->head == NULL) return 0;
  
    struct list_node *slow = list->head;
    struct list_node *fast = list->head;
  
    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
      
        if (slow == fast) {
            return 1;  // 有环
        }
    }
  
    return 0;  // 无环
}
```

### 3. 查找链表中间节点
```c
int linked_list_find_middle(const struct linked_list *list, int *data) {
    if (list == NULL || data == NULL || list->head == NULL) return 0;
  
    struct list_node *slow = list->head;
    struct list_node *fast = list->head;
  
    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
    }
  
    *data = slow->data;
    return 1;
}
```

### 4. 合并两个有序链表
```c
struct list_node* merge_sorted_lists(struct list_node *l1, struct list_node *l2) {
    struct list_node dummy;
    struct list_node *current = &dummy;
  
    while (l1 != NULL && l2 != NULL) {
        if (l1->data <= l2->data) {
            current->next = l1;
            l1 = l1->next;
        } else {
            current->next = l2;
            l2 = l2->next;
        }
        current = current->next;
    }
  
    // 连接剩余节点
    current->next = (l1 != NULL) ? l1 : l2;
  
    return dummy.next;
}
```

### 5. LRU缓存实现
```c
#define LRU_CACHE_SIZE 5

struct lru_cache {
    struct doubly_linked_list list;
    int capacity;
};

void lru_cache_init(struct lru_cache *cache) {
    doubly_linked_list_init(&cache->list);
    cache->capacity = LRU_CACHE_SIZE;
}

int lru_cache_get(struct lru_cache *cache, int key) {
    // 在实际实现中，这里应该是查找哈希表
    // 简化版本直接在链表中查找
    struct dlist_node *current = cache->list.head;
    while (current != NULL) {
        if (current->data == key) {
            // 将访问的节点移动到头部
            if (current != cache->list.head) {
                // 从当前位置删除
                if (current == cache->list.tail) {
                    cache->list.tail = current->prev;
                } else {
                    current->next->prev = current->prev;
                }
                current->prev->next = current->next;
              
                // 插入到头部
                current->next = cache->list.head;
                current->prev = NULL;
                cache->list.head->prev = current;
                cache->list.head = current;
            }
            return key;
        }
        current = current->next;
    }
    return -1;  // 未找到
}

void lru_cache_put(struct lru_cache *cache, int key) {
    // 检查是否已存在
    if (lru_cache_get(cache, key) != -1) {
        return;  // 已存在，get操作已将其移动到头部
    }
  
    // 如果缓存已满，删除尾部节点
    if (cache->list.size >= cache->capacity) {
        int temp;
        doubly_linked_list_pop_back(&cache->list, &temp);
    }
  
    // 在头部插入新节点
    doubly_linked_list_push_front(&cache->list, key);
}
```

## 性能分析

### 时间复杂度
| 操作 | 单向链表 | 双向链表 | 循环链表 |
|------|----------|----------|----------|
| 头部插入 | O(1) | O(1) | O(1) |
| 尾部插入 | O(n) | O(1) | O(n) |
| 中间插入 | O(n) | O(n) | O(n) |
| 头部删除 | O(1) | O(1) | O(1) |
| 尾部删除 | O(n) | O(1) | O(n) |
| 中间删除 | O(n) | O(n) | O(n) |
| 查找 | O(n) | O(n) | O(n) |
| 访问指定位置 | O(n) | O(n) | O(n) |

### 空间复杂度
- **单向链表**：O(n)，每个节点额外需要一个指针
- **双向链表**：O(n)，每个节点额外需要两个指针
- **循环链表**：O(n)，与普通链表相同

## 三种链表实现对比

| 特性 | 单向链表 | 双向链表 | 循环链表 |
|------|----------|----------|----------|
| 内存开销 | 最小 | 较大（双指针） | 与单向链表相同 |
| 遍历方向 | 只能向前 | 前后均可 | 从前到后（循环） |
| 删除操作 | 需要前驱节点 | 直接删除 | 需要找到前驱 |
| 实现复杂度 | 简单 | 中等 | 复杂 |
| 应用场景 | 一般链表操作 | 需要前后遍历 | 需要循环访问 |

### 注意事项
1. **内存管理**：始终记得释放删除节点的内存
2. **空指针检查**：操作前检查指针是否为空
3. **循环链表边界**：遍历时需要特殊处理终止条件
4. **双向链表指针**：插入删除时要同时维护prev和next指针
5. **性能考虑**：根据应用需求选择合适的链表类型
