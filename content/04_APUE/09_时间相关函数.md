# 时间相关函数详解

## 概述
时间相关函数是C语言标准库中处理时间数据的重要工具，主要包含时间获取、时间格式转换、时间计算和延时等功能，广泛应用于嵌入式系统的时间管理、日志记录、定时任务等场景。

## 基本时间获取

### time() 函数
获取当前系统时间戳（从1970年1月1日00:00:00 UTC开始的秒数）。

```c
#include <time.h>
#include <stdio.h>

time_t time(time_t *tloc);

// 使用示例
int main() {
    time_t current_time;
  
    // 方式1：传入NULL，只返回时间戳
    current_time = time(NULL);
    printf("Current timestamp: %ld\n", current_time);
  
    // 方式2：传入指针，同时存储到变量中
    time(&current_time);
    printf("Current timestamp: %ld\n", current_time);
  
    return 0;
}
```

**参数说明：**
- `tloc`: 指向time_t变量的指针，如果非NULL，则同时将结果存储到该位置
- **返回值**: 成功返回时间戳，失败返回-1

## 时间转换函数

```c
#include <time.h>

char *asctime(const struct tm *tm);                    // struct tm -> 字符串
char *asctime_r(const struct tm *tm, char *buf);       // 可重入版本
char *ctime(const time_t *timep);                      // time_t -> 字符串
char *ctime_r(const time_t *timep, char *buf);         // 可重入版本
```

### 示例用法
```c
#include <stdio.h>
#include <time.h>

int main() {
    time_t now;
    struct tm *local_time;
    
    time(&now);
    local_time = localtime(&now);
    
    printf("asctime格式: %s", asctime(local_time));
    printf("ctime格式: %s", ctime(&now));
    
    return 0;
}
```

### gmtime() 和 gmtime_r() - UTC时间转换

```c
#include <time.h>
#include <stdio.h>

struct tm *gmtime(const time_t *timep);
struct tm *gmtime_r(const time_t *timep, struct tm *result);

// 使用示例
int main() {
    time_t rawtime;
    struct tm *utc_time;
    struct tm utc_time_r;
  
    time(&rawtime);
  
    // 线程不安全版本
    utc_time = gmtime(&rawtime);
    printf("UTC Time: %04d-%02d-%02d %02d:%02d:%02d\n",
           utc_time->tm_year + 1900,
           utc_time->tm_mon + 1,
           utc_time->tm_mday,
           utc_time->tm_hour,
           utc_time->tm_min,
           utc_time->tm_sec);
  
    // 线程安全版本
    gmtime_r(&rawtime, &utc_time_r);
    printf("UTC Time (safe): %04d-%02d-%02d %02d:%02d:%02d\n",
           utc_time_r.tm_year + 1900,
           utc_time_r.tm_mon + 1,
           utc_time_r.tm_mday,
           utc_time_r.tm_hour,
           utc_time_r.tm_min,
           utc_time_r.tm_sec);
  
    return 0;
}
```

### mktime() - 时间结构体转时间戳

```c
#include <time.h>
#include <stdio.h>

time_t mktime(struct tm *tm);

// 使用示例
int main() {
    struct tm time_info = {0};
    time_t timestamp;
  
    // 设置特定时间：2025年1月1日 12:30:45
    time_info.tm_year = 2025 - 1900;  // 年份要减1900
    time_info.tm_mon = 1 - 1;         // 月份要减1 (0-11)
    time_info.tm_mday = 1;
    time_info.tm_hour = 12;
    time_info.tm_min = 30;
    time_info.tm_sec = 45;
  
    timestamp = mktime(&time_info);
    if (timestamp != -1) {
        printf("Timestamp for 2025-01-01 12:30:45: %ld\n", timestamp);
    } else {
        printf("Failed to convert time\n");
    }
  
    return 0;
}
```

## 时间结构体详解

### struct tm 结构详细说明

```c
struct tm {
    int tm_sec;    /* 秒 (0-60) - 允许闰秒 */
    int tm_min;    /* 分 (0-59) */
    int tm_hour;   /* 时 (0-23) - 24小时制 */
    int tm_mday;   /* 月中的天 (1-31) */
    int tm_mon;    /* 月份 (0-11) - 0表示1月 */
    int tm_year;   /* 年份 - 1900 */
    int tm_wday;   /* 星期几 (0-6, Sunday = 0) */
    int tm_yday;   /* 年中的天 (0-365, 1 Jan = 0) */
    int tm_isdst;  /* 夏令时标志 (-1/0/1) */
};

// 完整示例：解析和显示时间信息
void display_time_info(time_t timestamp) {
    struct tm *time_info = localtime(&timestamp);
  
    const char *weekdays[] = {
        "Sunday", "Monday", "Tuesday", "Wednesday",
        "Thursday", "Friday", "Saturday"
    };
  
    const char *months[] = {
        "January", "February", "March", "April",
        "May", "June", "July", "August",
        "September", "October", "November", "December"
    };
  
    printf("Date and Time Information:\n");
    printf("  Year: %d\n", time_info->tm_year + 1900);
    printf("  Month: %s (%d)\n", months[time_info->tm_mon], time_info->tm_mon + 1);
    printf("  Day: %d\n", time_info->tm_mday);
    printf("  Weekday: %s\n", weekdays[time_info->tm_wday]);
    printf("  Time: %02d:%02d:%02d\n", 
           time_info->tm_hour, time_info->tm_min, time_info->tm_sec);
    printf("  Day of Year: %d\n", time_info->tm_yday + 1);
    printf("  DST: %s\n", 
           time_info->tm_isdst > 0 ? "Yes" : 
           time_info->tm_isdst == 0 ? "No" : "Unknown");
}
```

## 格式化时间输出

### strftime() - 格式化时间字符串

```c
#include <time.h>
#include <stdio.h>

size_t strftime(char *s, size_t max, const char *format, const struct tm *tm);

// 使用示例
int main() {
    time_t rawtime;
    struct tm *timeinfo;
    char buffer[80];
  
    time(&rawtime);
    timeinfo = localtime(&rawtime);
  
    // 不同格式的时间输出
    strftime(buffer, sizeof(buffer), "%Y-%m-%d %H:%M:%S", timeinfo);
    printf("Standard format: %s\n", buffer);
  
    strftime(buffer, sizeof(buffer), "%A, %B %d, %Y", timeinfo);
    printf("Full format: %s\n", buffer);
  
    strftime(buffer, sizeof(buffer), "%c", timeinfo);
    printf("Locale format: %s\n", buffer);
  
    return 0;
}

// 常用格式化符号说明
/*
%Y - 四位年份 (如: 2025)
%y - 两位年份 (如: 25)
%m - 月份 (01-12)
%d - 日期 (01-31)
%H - 小时 (00-23)
%M - 分钟 (00-59)
%S - 秒 (00-60)
%A - 完整星期名 (如: Monday)
%a - 简写星期名 (如: Mon)
%B - 完整月份名 (如: January)
%b - 简写月份名 (如: Jan)
%c - 本地日期时间表示
%x - 本地日期表示
%X - 本地时间表示
*/
```

## 高精度时间函数

### gettimeofday() - 微秒级时间

```c
#include <sys/time.h>
#include <stdio.h>

int gettimeofday(struct timeval *tv, struct timezone *tz);

// 使用示例
int main() {
    struct timeval tv;
  
    if (gettimeofday(&tv, NULL) == 0) {
        printf("Seconds: %ld\n", tv.tv_sec);
        printf("Microseconds: %ld\n", tv.tv_usec);
        printf("Full time: %ld.%06ld\n", tv.tv_sec, tv.tv_usec);
    }
  
    return 0;
}

// 精确延时函数示例
void precise_delay_us(long microseconds) {
    struct timeval start, current;
    gettimeofday(&start, NULL);
  
    do {
        gettimeofday(&current, NULL);
    } while ((current.tv_sec - start.tv_sec) * 1000000 + 
             (current.tv_usec - start.tv_usec) < microseconds);
}
```

## sleep函数详解

### sleep() - 秒级休眠

```c
#include <unistd.h>
#include <stdio.h>

unsigned int sleep(unsigned int seconds);

// 使用示例
int main() {
    printf("Sleeping for 3 seconds...\n");
    unsigned int remaining = sleep(3);
    if (remaining > 0) {
        printf("Sleep interrupted with %u seconds remaining\n", remaining);
    } else {
        printf("Sleep completed\n");
    }
  
    return 0;
}
```

### usleep() - 微秒级休眠

```c
#include <unistd.h>
#include <stdio.h>

int usleep(useconds_t usec);

// 使用示例
int main() {
    printf("Sleeping for 500000 microseconds (0.5 seconds)...\n");
    if (usleep(500000) == 0) {
        printf("Microsleep completed\n");
    } else {
        perror("usleep failed");
    }
  
    return 0;
}
```

### nanosleep() - 纳秒级休眠（推荐）

```c
#include <time.h>
#include <stdio.h>

int nanosleep(const struct timespec *req, struct timespec *rem);

// 使用示例
int main() {
    struct timespec req, rem;
  
    // 休眠1.5秒
    req.tv_sec = 1;
    req.tv_nsec = 500000000;  // 500毫秒 = 500,000,000纳秒
  
    if (nanosleep(&req, &rem) == -1) {
        printf("Sleep interrupted\n");
        if (rem.tv_sec > 0 || rem.tv_nsec > 0) {
            printf("Remaining time: %ld.%09ld seconds\n", 
                   rem.tv_sec, rem.tv_nsec);
        }
    } else {
        printf("Nanosleep completed\n");
    }
  
    return 0;
}
```

## 实用工具函数封装

### 时间工具类封装示例

```c
#include <time.h>
#include <sys/time.h>
#include <stdio.h>
#include <string.h>

// 获取格式化时间字符串
void get_formatted_time(char *buffer, size_t size, const char *format) {
    time_t rawtime;
    struct tm *timeinfo;
  
    time(&rawtime);
    timeinfo = localtime(&rawtime);
    strftime(buffer, size, format, timeinfo);
}

// 获取毫秒时间戳
long long get_millisecond_timestamp() {
    struct timeval tv;
    gettimeofday(&tv, NULL);
    return (long long)tv.tv_sec * 1000 + tv.tv_usec / 1000;
}

// 计算时间差（毫秒）
long long time_diff_ms(struct timeval start, struct timeval end) {
    return (end.tv_sec - start.tv_sec) * 1000 + 
           (end.tv_usec - start.tv_usec) / 1000;
}

// 性能测试示例
void performance_test_example() {
    struct timeval start, end;
  
    gettimeofday(&start, NULL);
  
    // 执行一些操作
    for (int i = 0; i < 1000000; i++) {
        // 模拟工作负载
        volatile int x = i * i;
    }
  
    gettimeofday(&end, NULL);
  
    long long duration = time_diff_ms(start, end);
    printf("Operation took %lld milliseconds\n", duration);
}

// 完整的时间信息显示
void display_complete_time_info() {
    time_t rawtime;
    struct tm *timeinfo;
    char buffer[100];
  
    time(&rawtime);
    timeinfo = localtime(&rawtime);
  
    printf("=== 时间信息 ===\n");
    printf("时间戳: %ld\n", rawtime);
  
    get_formatted_time(buffer, sizeof(buffer), "%Y-%m-%d %H:%M:%S");
    printf("标准时间: %s\n", buffer);
  
    get_formatted_time(buffer, sizeof(buffer), "%A, %B %d, %Y");
    printf("完整日期: %s\n", buffer);
  
    printf("星期: %d (0=Sunday)\n", timeinfo->tm_wday);
    printf("年内第几天: %d\n", timeinfo->tm_yday + 1);
    printf("夏令时: %s\n", timeinfo->tm_isdst ? "是" : "否");
  
    printf("毫秒时间戳: %lld\n", get_millisecond_timestamp());
    printf("===============\n");
}
```

## 注意事项和最佳实践

### 1. 线程安全性
- `gmtime()` 和 `localtime()` 不是线程安全的
- 推荐使用 `_r` 后缀的线程安全版本

### 2. 时区处理
```c
// 设置时区
setenv("TZ", "Asia/Shanghai", 1);
tzset();  // 应用时区设置
```

### 3. 错误处理
- 检查所有时间函数的返回值
- `mktime()` 失败时返回 -1
- `time()` 失败时返回 -1

### 4. 跨平台兼容性
- `usleep()` 在某些系统上只支持最大1秒的延时
- `nanosleep()` 是更现代和可移植的选择

### 5. 嵌入式系统特殊考虑
- 某些嵌入式系统RTC时钟可能没有夏令时支持
- 实时时钟可能需要校准
- 考虑使用硬件定时器而非软件延时
