## 实用工具函数

### 输入输出重定向

Linux/Unix系统中I/O重定向是一种强大机制，通过文件描述符（File Descriptor）实现灵活的输入输出控制。每个进程都有预定义的标准输入（stdin）、标准输出（stdout）和标准错误（stderr）。

#### 基础语法总结：

```shell
# I/O重定向
命令 < 文件                    # 标准输入重定向 (fd 0) - 从文件读取输入
命令 > 文件                    # 标准输出重定向 (fd 1) - 覆盖写入文件
命令 2> 文件                   # 标准错误输出重定向 (fd 2) - 错误信息写入文件

# 追加模式
命令 >> 文件                   # 标准输出追加到文件末尾
命令 2>> 文件                  # 标准错误追加到文件末尾

# 合并输出（常用于日志记录）
命令 &> 文件                   # bash特有语法，同时覆盖 stdout 和 stderr
命令 > 文件 2>&1               # stdout先重定向到文件，再将stderr指向stdout
命令 2>&1 > 文件               # 注意执行顺序：先错误合并stdout，后stdout转向文件（可能不生效）

# 文件描述符操作
命令 3> 文件                   # 打开文件描述符3进行写入
命令 3<&0                      # 将文件描述符3重定向为标准输入
命令 3>&-                      # 关闭文件描述符3

```

#### 实际应用实例

```bash
# 编译程序并将所有输出重定向到日志
gcc program.c > compile.log 2>&1

# 使用input.txt内容运行程序，输出结果保存至output.txt
./program < input.txt > output.txt

# 忽略权限错误地查找所有包含“error”的日志
grep "error" /var/log/*.log 2>/dev/null

# 混合使用多个文件描述符
exec 3< input.data       # 打开输入文件
while read -u 3 line; do # 从fd=3中读取
    echo $line
done
exec 3<&-                # 关闭描述符
```

#### I/O描述符表

| 文件描述符 | 名称         | 默认设备 | 用途说明         |
|------------|--------------|----------|------------------|
| 0          | stdin        | 键盘     | 标准输入         |
| 1          | stdout       | 屏幕     | 标准输出         |
| 2          | stderr       | 屏幕     | 标准错误输出     |
| 3-9        | 自定义fd     | 需显式打开| 用户自定义使用   |

### 特殊设备文件

Linux 提供了一些虚拟设备节点，可以模拟特定的数据行为，是高级脚本和底层操作中不可或缺的资源。

#### `/dev/null` - 数据黑洞

**作用**：任何写入该设备的数据都会被丢弃，读取则返回EOF（空）。
**常用场景**：
- 隐藏程序输出
- 实现“吃掉”不需要的输出，防止垃圾信息污染终端

```bash
# 示例
echo "hello world" > /dev/null     # 输出被完全忽略
rm -rf /nonexistent_dir 2>/dev/null # 忽略删除失败的信息

# C语言模拟效果（不推荐实际使用）
freopen("/dev/null", "w", stdout); // 所有printf不会显示
```

#### `/dev/zero` - 零字节生成器

**作用**：提供无限个 `\0` 字节流供消费。

**应用示例**：
```bash
# 创建一个大小固定的全零文件（比如做测试镜像）
dd if=/dev/zero of=test.bin bs=1K count=100   # 生成100KB零填充文件

# 性能测试：模拟大文件写入
dd if=/dev/zero of=/mnt/ramdisk/testfile bs=1M count=10 oflag=direct
```

#### `/dev/random` 与 `/dev/urandom` - 随机数源

| 设备        | 说明                              | 是否阻塞 |
|-------------|-----------------------------------|----------|
| `/dev/random`   | 获取高质量真随机数（内核熵池） | 是       |
| `/dev/urandom`  | 伪随机数（无需等待熵池）           | 否       |

**应用示例**：
```bash
# 生成加密密钥种子材料
dd if=/dev/urandom bs=1 count=32 > keyfile.bin

# Shell生成临时密码
cat /dev/urandom | tr -dc 'a-zA-Z0-9' | fold -w 16 | head -n 1
```

### 文件操作工具函数（C语言封装版）

提供常用文件操作函数，方便在嵌入式环境中安全读取、写入和检查文件属性。

#### 代码展示

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>

// 安全读取文件
int safe_file_read(const char* filename, char** buffer, size_t* size) {
    FILE* fp = fopen(filename, "rb");
    if (!fp) return -1;

    fseek(fp, 0, SEEK_END);
    long length = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    *buffer = calloc(length + 1, sizeof(char));
    if (!*buffer) {
        fclose(fp);
        return -2;
    }

    *size = fread(*buffer, 1, length, fp);
    fclose(fp);
    return 0;
}

// 安全写入文件
int safe_file_write(const char* filename, const char* data, size_t len) {
    FILE* fp = fopen(filename, "wb");
    if (!fp) return -1;

    size_t written = fwrite(data, 1, len, fp);
    fclose(fp);

    return (written == len) ? 0 : -2;
}

// 检查文件是否存在
static inline int file_exists(const char* path) {
    return access(path, F_OK) == 0;
}

// 获取文件大小
static off_t get_filesize(const char* path) {
    struct stat st;
    return (stat(path, &st) == 0) ? st.st_size : -1;
}
```

### 字符串处理工具函数（C语言）

用于字符串清理、分割及格式转换，支持各种常见需求。

#### 示例代码

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

// 移除字符串前后空白字符
char* trim(char* str) {
    char* end;

    while (isspace((unsigned char)*str)) str++; // 去头
    if (*str == 0) return str;

    end = str + strlen(str) - 1;
    while (end > str && isspace((unsigned char)*end)) end--;
    *(end + 1) = '\0';
    return str;
}

// 分割字符串
char** split_string(const char* s, char delimiter, int* count) {
    char* temp = strdup(s);
    char* token = strtok(temp, &delimiter);
    char** tokens = NULL;
    int num = 0;

    while (token != NULL) {
        tokens = realloc(tokens, sizeof(char*) * (num + 1));
        tokens[num++] = strdup(token);
        token = strtok(NULL, &delimiter);
    }

    free(temp);
    *count = num;
    return tokens;
}

// 十六进制 → 字节数组
int hex2bytes(const char* hex, uint8_t* bytes, int maxlen) {
    if (strlen(hex) % 2) return -1;
    int n = strlen(hex) / 2;
    if (n > maxlen) return -1;

    for (int i = 0; i < n; i++) {
        sscanf(hex + i * 2, "%2hhx", &bytes[i]);
    }
    return n;
}

// 字节数组 → 十六进制字符串
void bytes2hex(const uint8_t* data, int len, char* hex_str) {
    for (int i = 0; i < len; i++) {
        sprintf(hex_str + i * 2, "%02x", data[i]);
    }
    hex_str[len * 2] = '\0'; // Null terminate
}
```

### 系统信息获取工具函数

提供运行时间、负载、版本信息等系统状态查询接口。

#### 示例代码

```c
#include <sys/utsname.h>
#include <sys/sysinfo.h>
#include <time.h>
#include <stdio.h>

// 获取系统运行时间（秒）
long get_uptime() {
    struct sysinfo info;
    if (sysinfo(&info) == 0)
        return info.uptime;
    return -1;
}

// 获取当前平均负载：过去1/5/15分钟
int get_avg_load(double avg[3]) {
    return getloadavg(avg, 3);
}

// 获取当前本地时间并格式化为"YYYY-MM-DD HH:MM:SS"
void get_localtime_str(char* out_buf, size_t buf_len) {
    time_t rawtime;
    struct tm* timeinfo;

    time(&rawtime);
    timeinfo = localtime(&rawtime);

    strftime(out_buf, buf_len, "%Y-%m-%d %H:%M:%S", timeinfo);
}

// 显示系统基本信息
void print_system_info() {
    struct utsname info;
    if (uname(&info) != 0) return;

    printf("System name : %s\n", info.sysname);
    printf("Node        : %s\n", info.nodename);
    printf("Release     : %s\n", info.release);
    printf("Version     : %s\n", info.version);
    printf("Machine     : %s\n", info.machine);
}
```

### 内存操作工具函数

主要用于安全内存操作，防止溢出或泄露敏感信息。

#### 示例代码

```c
#include <string.h>
#include <stdint.h>

// 不检查 NULL 安全地复制内存
void* safe_memmove(void* dest, const void* src, size_t count) {
    if (!dest || !src || count == 0)
        return NULL;
    return memmove(dest, src, count);
}

// 清空内存以防止残留敏感信息（volatile防编译优化）
void secure_zero_memory(void* buf, size_t len) {
    volatile uint8_t* ptr = (uint8_t*)buf;
    while (len--) *ptr++ = 0;
}

// 常量时间比较：避免基于比较时间侧信道攻击
int secure_mem_cmp(const void* a, const void* b, size_t len) {
    const uint8_t* u1 = (const uint8_t*)a;
    const uint8_t* u2 = (const uint8_t*)b;
    register uint8_t diff = 0;

    for (size_t i = 0; i < len; i++) {
        diff |= u1[i] ^ u2[i];
    }
    return (diff == 0) ? 0 : -1;
}
```

### 错误处理工具函数

简化错误码转换和日志记录，帮助调试与分析问题。

#### 示例代码

```c
#include <errno.h>
#include <stdarg.h>
#include <time.h>
#include <stdio.h>

// 将错误码转为人类可读字符串
const char* error_string(int errnum) {
    static char buf[256];
#if defined(_GNU_SOURCE)
    return strerror_r(errnum, buf, sizeof(buf));
#else
    strerror_r(errnum, buf, sizeof(buf));
    return buf;
#endif
}

// 标准日志打印函数
void log_printf(const char* level, const char* fmt, ...) {
    char ts[32];
    time_t t = time(NULL);
    struct tm *tm_info = localtime(&t);

    strftime(ts, sizeof(ts), "%F %T", tm_info);

    va_list args;
    va_start(args, fmt);
    fprintf(stderr, "[%s][%s] ", ts, level);
    vfprintf(stderr, fmt, args);
    va_end(args);
    fprintf(stderr, "\n");
}
```

### 网络地址操作工具函数

适用于基础网络编程场景，处理IP校验及字节序转换。

#### 示例代码

```c
#include <netinet/in.h>
#include <arpa/inet.h>

// 检查IPv4合法性
int is_ipv4_valid(const char* addr_str) {
    struct in_addr addr;
    return inet_pton(AF_INET, addr_str, &addr) == 1;
}

// 验证端口号是否合法（1～65535）
inline int is_port_valid(int port) {
    return port > 0 && port <= 65535;
}

// 网络字节序便捷宏定义
uint16_t hton16(uint16_t x) { return htons(x); }
uint32_t hton32(uint32_t x) { return htonl(x); }
uint16_t ntoh16(uint16_t x) { return ntohs(x); }
uint32_t ntoh32(uint32_t x) { return ntohl(x); }
```
